03. Caesar Cipher

JAVA
JAVA SOURCE CODE 
import java.util.Scanner;

public class CaesarCipher {

    public static String encrypt(String text, int key) {
        String result = "";

        for (int i = 0; i < text.length(); i++) {
            char ch = text.charAt(i);

            if (ch >= 'A' && ch <= 'Z') {
                ch = (char) ('A' + (ch - 'A' + key) % 26);
            } else if (ch >= 'a' && ch <= 'z') {
                ch = (char) ('a' + (ch - 'a' + key) % 26);
            }

            result += ch;
        }

        return result;
    }

    public static String decrypt(String text, int key) {
        return encrypt(text, (26 - key % 26) % 26);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter plaintext: ");
        String text = sc.nextLine();

        System.out.print("Enter key (shift): ");
        int key = sc.nextInt();
        key = ((key % 26) + 26) % 26;
        String encrypted = encrypt(text, key);
        String decrypted = decrypt(encrypted, key);

        System.out.println("Encrypted text: " + encrypted);
        System.out.println("Decrypted text: " + decrypted);

        sc.close();
    }
}

04. Substitution Cipher

import java.util.*;

public class SubstitutionCipher {
    public static String crypto(String str, String encrypted, String a, String b) {
        char c;
        for (int i = 0; i < str.length(); i++) {
            c = str.charAt(i);
            int j = a.indexOf(c);
            if (j != -1)
                encrypted = encrypted + b.charAt(j);
            else
                encrypted = encrypted + c;
        }
        return encrypted;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String a = "abcdefghijklmnopqrstuvwxyz";
        String b = "zyxwvutsrqponmlkjihgfedcba";
        System.out.print("Enter any string: ");
        String str = sc.nextLine();
        String encrypted = "";
        String decrypted = "";
        encrypted = crypto(str, "", a, b);
        decrypted = crypto(encrypted, "", b, a);

        System.out.println("The encrypted data is: " + encrypted);
        System.out.println("The decrypted data is: " + decrypted);
        sc.close();
    }
}

import java.util.*;

class HillCipher {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int k[][] = new int[3][3];
        System.out.println("Enter 3x3 matrix:");
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                k[i][j] = sc.nextInt();
            }
        }
        System.out.println("Enter 3-letter text");
        String s = sc.next().toLowerCase();
        int x[] = new int[3];
        int y[] = new int[3];

        // Convert characters to numbers
        for (int i = 0; i < 3; i++) {
            x[i] = s.charAt(i) - 'a';
        }

        // Encryption
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                y[i] += k[i][j] * x[j];
            }
            y[i] = y[i] % 26;
        }

        System.out.println("Encrypted:");
        for (int i = 0; i < 3; i++) {
            System.out.print((char) (y[i] + 'a'));
        }
        System.out.println();
        sc.close();
    }
}

06. DES Algorithm

JAVA
JAVA SOURCE CODE 
import java.util.*;
import javax.crypto.*;

public class DES {
    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter text:");
        String s = sc.nextLine();

        // Generate DES key
        KeyGenerator kg = KeyGenerator.getInstance("DES");
        SecretKey key = kg.generateKey();

        // Create cipher
        Cipher c = Cipher.getInstance("DES");

        // Encryption
        c.init(Cipher.ENCRYPT_MODE, key);
        byte[] enc = c.doFinal(s.getBytes());
        System.out.println("Encrypted:" + Base64.getEncoder().encodeToString(enc));

        // Decryption
        c.init(Cipher.DECRYPT_MODE, key);
        byte[] dec = c.doFinal(enc);
        System.out.println("Decrypted:" + new String(dec));
        sc.close();
    }
}

09. RSA Algorithm

JAVA
JAVA SOURCE CODE 
import java.math.BigInteger;

public class RSA {
    public static void main(String[] args) {
        BigInteger p = BigInteger.valueOf(61);
        BigInteger q = BigInteger.valueOf(53);
        BigInteger n = p.multiply(q);
        BigInteger phi = (p.subtract(BigInteger.ONE)).multiply(q.subtract(BigInteger.ONE));
        BigInteger e = BigInteger.valueOf(17);
        BigInteger d = e.modInverse(phi);
        BigInteger msg = BigInteger.valueOf(65);

        // Encryption
        BigInteger enc = msg.modPow(e, n);

        // Decryption
        BigInteger dec = enc.modPow(d, n);

        System.out.println("Public Key: (" + e + "," + n + ")");
        System.out.println("Private Key:(" + d + "," + n + ")");
        System.out.println("Encrypted Message:" + enc);
        System.out.println("Decrypted Message:" + dec);
    }
}

10a. Diffie-Hellman Key Exchange

JAVA
JAVA SOURCE CODE 
import java.math.BigInteger;

public class DeffieHellman {
    public static void main(String[] args) {
        // Public values: prime p and base g
        BigInteger p = BigInteger.valueOf(47);
        BigInteger g = BigInteger.valueOf(71);

        // Private keys
        BigInteger A = BigInteger.valueOf(9);  // Alice private key
        BigInteger B = BigInteger.valueOf(14); // Bob private key

        // Calculate public keys
        BigInteger AP = g.modPow(A, p);
        BigInteger BP = g.modPow(B, p);

        // Calculate shared secret keys
        BigInteger AS = BP.modPow(A, p);
        BigInteger BS = AP.modPow(B, p);

        System.out.println("Alice Shared Key:" + AS);
        System.out.println("Bob Shared Key:" + BS);
    }
}

import java.util.*;
import java.security.*;

        System.out.print("Enter text:");
        String text = sc.nextLine();

        MessageDigest md = MessageDigest.getInstance("SHA-256");

13. Hill Cipher (Encryption & Decryption)

JAVA
JAVA SOURCE CODE 
import java.util.Scanner;

public class HillCipher {

    static int[][] key;
    static int[][] inverse;
    static int n;

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter matrix size (2 or 3): ");
        n = sc.nextInt();

        if (n != 2 && n != 3) {
            System.out.println("Only 2x2 and 3x3 are supported.");
            sc.close();
            return;
        }

        key = new int[n][n];
        inverse = new int[n][n];

        System.out.println("Enter " + n + "x" + n + " key matrix:");

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                key[i][j] = sc.nextInt();
            }
        }

        sc.nextLine();

        System.out.print("Enter message: ");
        String msg = sc.nextLine().toLowerCase();
        msg = msg.replaceAll("[^a-z]", "");

        if (msg.isEmpty()) {
            System.out.println("Enter a valid message.");
            sc.close();
            return;
        }

        // Pad message to a multiple of matrix size
        while (msg.length() % n != 0) {
            msg += "x";
        }

        if (!findInverse()) {
            System.out.println(
                "Key matrix is not invertible modulo 26.");
            sc.close();
            return;
        }

        String enc = process(msg, key);
        String dec = process(enc, inverse);

        System.out.println("Padded message: " + msg);
        System.out.println("Encrypted string: " + enc);
        System.out.println("Decrypted string: " + dec);

        sc.close();
    }

    static String process(String msg, int[][] matrix) {

        StringBuilder result = new StringBuilder();

        for (int start = 0; start < msg.length(); start += n) {

            for (int i = 0; i < n; i++) {

                int sum = 0;

                for (int j = 0; j < n; j++) {
                    int value = msg.charAt(start + j) - 'a';
                    sum += matrix[i][j] * value;
                }

                sum = mod(sum);
                result.append((char) (sum + 'a'));
            }
        }

        return result.toString();
    }

    static int mod(int x) {
        return ((x % 26) + 26) % 26;
    }

    // Determinant for 2x2 or 3x3 matrix
    static int determinant(int[][] a, int size) {

        if (size == 1) {
            return a[0][0];
        }

        if (size == 2) {
            return a[0][0] * a[1][1]
                 - a[0][1] * a[1][0];
        }

        return a[0][0] *
                   (a[1][1] * a[2][2]
                  - a[1][2] * a[2][1])
             - a[0][1] *
                   (a[1][0] * a[2][2]
                  - a[1][2] * a[2][0])
             + a[0][2] *
                   (a[1][0] * a[2][1]
                  - a[1][1] * a[2][0]);
    }

    static int modInverse(int value) {
        for (int i = 1; i < 26; i++) {
            if (mod(value * i) == 1) {
                return i;
            }
        }
        return -1;
    }

    // Create a minor matrix by removing one row and one column
    static int[][] minorMatrix(int[][] a, int row, int col) {

        int[][] minor = new int[n - 1][n - 1];
        int r = 0;

        for (int i = 0; i < n; i++) {
            if (i == row) continue;

            int c = 0;

            for (int j = 0; j < n; j++) {
                if (j == col) continue;

                minor[r][c] = a[i][j];
                c++;
            }

            r++;
        }

        return minor;
    }

    static boolean findInverse() {

        int det = mod(determinant(key, n));
        int detInv = modInverse(det);

        if (detInv == -1) {
            return false;
        }

        int[][] adj = new int[n][n];

        // Calculate cofactors and transpose them
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {

                int[][] minor = minorMatrix(key, i, j);
                int cofactor = determinant(minor, n - 1);

                if ((i + j) % 2 != 0) {
                    cofactor = -cofactor;
                }

                adj[j][i] = mod(cofactor);
            }
        }

        // Inverse = determinant inverse * adjugate
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                inverse[i][j] = mod(adj[i][j] * detInv);
            }
        }

        return true;
    }
}
